// Grid HUD & Logistics — Refineries v2: refineries 24/7 on the server (rafinerie 24/7 na serwerze).
// Same refinery rules as players know from Isy's Inventory Manager (own implementation)
// (z poprawką „każda ruda po równo”), bez reszty Isy:
//  1. lista rud: najpierw kolejność z priority=, potem rudy, których sztabek jest najmniej
//  2. pełna rafineria (98%) bez pierwszej dostępnej rudy z listy oddaje wszystko do magazynu
//  3. każda ruda wg listy po równo: ceil(suma / liczba rafinerii), nadmiar > 10% wraca
//  4. pierwsza ruda z listy, którą rafineria ma, idzie na początek kolejki
//  + sztabki z wyjść rafinerii do kontenerów Ingots (przy wyłączonym konwejerze gra ich nie wypycha)
// Settings in CustomData (written by the plugin: /data » Logistics):
//   priority=Uranium, Stone   ores=Ores   ingots=Ingots   skip=!manual, Locked

const string Tag = "[DaneHud-Rafinerie]";
string oreKw = "Ores", ingotKw = "Ingots";
List<string> prio = new List<string>(), skip = new List<string> { "!manual", "Locked" };
List<IMyRefinery> refs = new List<IMyRefinery>();
List<IMyTerminalBlock> cargo = new List<IMyTerminalBlock>(), ores = new List<IMyTerminalBlock>(), ingots = new List<IMyTerminalBlock>();
List<MyInventoryItem> items = new List<MyInventoryItem>();
List<MyItemType> allOres = new List<MyItemType>();
int run, moves, budget;
string current = "-";

public Program() { Runtime.UpdateFrequency = UpdateFrequency.Update100; }

List<string> Split(string v)
{
    var l = new List<string>();
    foreach (var s in v.Split(',')) if (s.Trim().Length > 0) l.Add(s.Trim());
    return l;
}

bool Skip(IMyTerminalBlock b)
{
    foreach (var k in skip) if (b.CustomName.Contains(k)) return true;
    return !b.IsFunctional || !b.IsSameConstructAs(Me);
}

void Scan()
{
    foreach (var raw in Me.CustomData.Split('\n'))
    {
        var l = raw.Trim(); int e = l.IndexOf('=');
        if (e < 0) continue;
        string k = l.Substring(0, e).Trim().ToLower(), v = l.Substring(e + 1).Trim();
        if (k == "priority") prio = Split(v);
        else if (k == "ores") oreKw = v;
        else if (k == "ingots") ingotKw = v;
        else if (k == "skip") skip = Split(v);
    }
    GridTerminalSystem.GetBlocksOfType(refs, r => !Skip(r) && r.Enabled);
    GridTerminalSystem.GetBlocksOfType<IMyCargoContainer>(cargo, c => !Skip(c));
    ores.Clear(); ingots.Clear();
    foreach (var c in cargo)
    {
        if (oreKw.Length > 0 && c.CustomName.Contains(oreKw)) ores.Add(c);
        if (ingotKw.Length > 0 && c.CustomName.Contains(ingotKw)) ingots.Add(c);
    }
    if (ores.Count == 0) ores.AddRange(cargo);
    if (ingots.Count == 0) ingots.AddRange(cargo);
    // rudy, które przyjmują rafinerie (wszystkie rudy, jakie widać w ładowniach i rafineriach, + z listy)
    foreach (var c in cargo) { items.Clear(); c.GetInventory(0).GetItems(items); foreach (var it in items) if (it.Type.TypeId == "MyObjectBuilder_Ore" && !allOres.Contains(it.Type)) allOres.Add(it.Type); }
    foreach (var p in prio) { var t = MyItemType.MakeOre(p); if (!allOres.Contains(t)) allOres.Add(t); }
}

bool HasSpace(IMyInventory i) => (float)i.CurrentVolume < (float)i.MaxVolume * 0.98f;

double Move(IMyInventory src, MyInventoryItem it, IMyInventory dst, double amount)
{
    if (budget <= 0 || amount <= 0) return 0;
    if (!src.CanTransferItemTo(dst, it.Type)) return 0;
    double a = Math.Min(amount, (double)it.Amount);
    if (src.TransferItemTo(dst, it, (MyFixedPoint)a)) { budget--; moves++; return a; }
    return 0;
}

IMyInventory FreeOre() { foreach (var c in ores) if (HasSpace(c.GetInventory(0))) return c.GetInventory(0); return null; }

public void Main(string arg, UpdateType src)
{
    if (!Me.CustomData.Contains(Tag)) Me.CustomData = Tag + "\npriority=Stone, Platinum\nores=Ores\ningots=Ingots\nskip=!manual, Locked\n" + Me.CustomData;
    if (run++ % 10 == 0) Scan();
    budget = 12;
    foreach (var r in refs) if (r.UseConveyorSystem) r.UseConveyorSystem = false;

    // sztabki z wyjść rafinerii do magazynu
    foreach (var r in refs)
    {
        var o = r.GetInventory(1);
        items.Clear(); o.GetItems(items);
        foreach (var it in items)
            foreach (var c in ingots)
                if (HasSpace(c.GetInventory(0)) && Move(o, it, c.GetInventory(0), (double)it.Amount) > 0) break;
    }
    if (refs.Count == 0) { Status(); return; }

    // ile jest rudy (magazyn + rafinerie) i sztabek
    var total = new Dictionary<MyItemType, double>();
    var ingot = new Dictionary<string, double>();
    foreach (var c in cargo)
    {
        items.Clear(); c.GetInventory(0).GetItems(items);
        foreach (var it in items)
        {
            double t;
            if (it.Type.TypeId == "MyObjectBuilder_Ore" && ores.Contains(c)) { total.TryGetValue(it.Type, out t); total[it.Type] = t + (double)it.Amount; }
            else if (it.Type.TypeId == "MyObjectBuilder_Ingot") { ingot.TryGetValue(it.Type.SubtypeId, out t); ingot[it.Type.SubtypeId] = t + (double)it.Amount; }
        }
    }
    foreach (var r in refs) { items.Clear(); r.GetInventory(0).GetItems(items); foreach (var it in items) { double t; total.TryGetValue(it.Type, out t); total[it.Type] = t + (double)it.Amount; } }

    // 1. lista rud
    var order = new List<MyItemType>();
    foreach (var p in prio) { var t = MyItemType.MakeOre(p); if (!order.Contains(t)) order.Add(t); }
    foreach (var t in allOres.Where(x => !order.Contains(x)).OrderBy(x => { double v; ingot.TryGetValue(x.SubtypeId, out v); return v; })) order.Add(t);
    order = order.Where(t => refs.Any(r => r.GetInventory(0).CanItemsBeAdded(1, t) || (double)r.GetInventory(0).GetItemAmount(t) > 0)).ToList();
    Func<MyItemType, double> amt = t => { double v; total.TryGetValue(t, out v); return v; };

    // 2. pełna rafineria bez pierwszej dostępnej rudy oddaje wszystko
    MyItemType top = default(MyItemType);
    bool hasTop = false;
    foreach (var t in order) if (amt(t) > 100) { top = t; hasTop = true; break; }
    current = hasTop ? top.SubtypeId : "-";
    if (hasTop)
        foreach (var r in refs)
        {
            var inv = r.GetInventory(0);
            if (HasSpace(inv) || (double)inv.GetItemAmount(top) > 0) continue;
            items.Clear(); inv.GetItems(items);
            foreach (var it in items) { var dst = FreeOre(); if (dst == null) break; Move(inv, it, dst, (double)it.Amount); }
        }

    // 3. każda ruda wg listy po równo
    foreach (var t in order)
    {
        double sum = amt(t);
        if (sum <= 0) continue;
        double share = Math.Ceiling(sum / refs.Count);
        foreach (var r in refs)
        {
            var inv = r.GetInventory(0);
            double need = share - (double)inv.GetItemAmount(t);
            if (need < -1 - share * 0.1)
            {
                var it = inv.FindItem(t); var dst = FreeOre();
                if (it.HasValue && dst != null) Move(inv, it.Value, dst, -need);
                continue;
            }
            if (!HasSpace(inv)) continue;
            foreach (var c in ores)
            {
                if (need <= 1) break;
                var ci = c.GetInventory(0);
                var it = ci.FindItem(t);
                if (it.HasValue) need -= Move(ci, it.Value, inv, need);
            }
        }
    }

    // 4. pierwsza ruda z listy na początek kolejki
    foreach (var r in refs)
    {
        var inv = r.GetInventory(0);
        items.Clear(); inv.GetItems(items);
        if (items.Count < 2) continue;
        foreach (var t in order)
        {
            int idx = items.FindIndex(x => x.Type == t);
            if (idx < 0) continue;
            if (idx > 0) inv.TransferItemTo(inv, idx, 0, true);
            break;
        }
    }
    Status();
}

void Status()
{
    Echo("Grid HUD & Logistics: Refineries v2\nRefineries / Rafinerie: " + refs.Count + "\nKontenery rud: " + ores.Count + ", sztabek: " + ingots.Count +
         "\nNajpierw: " + current + "\nPrzeniesienia: " + moves);
}
